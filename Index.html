// lib/sms/africastalking.ts import AfricaTalking from 'africastalking'; const africastalking = AfricaTalking({ apiKey: process.env.AT_API_KEY!, username: process.env.AT_USERNAME!, }); export async function sendTransferSMS(phoneNumber: string, amount: number, reference: string) { const options = { to: `+233${phoneNumber.slice(1)}`, message: `Hilary's Bank: ₵${amount} has been sent to your mobile money wallet. Reference: ${reference}`, from: 'HILARYS', }; try { const result = await africastalking.SMS.send(options); console.log('SMS sent:', result); return { success: true }; } catch (error) { console.error('SMS failed:', error); return { success: false, error }; } }
